cmake_minimum_required(VERSION 3.20)
project(industry_ready_cpp_course LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_executable(sanity src/sanity/main.cpp)
add_executable(oop_lab src/exercises/oop_object_model.cpp)
add_executable(raii_lab src/exercises/raii_file_handle.cpp)
add_executable(lru_cache_lab src/exercises/lru_cache.cpp)
add_executable(thread_pool_lab src/exercises/thread_pool.cpp)
add_executable(order_book_lab src/exercises/order_book.cpp)
add_executable(dsa_disjoint_set_lab src/dsa/disjoint_set.cpp)
add_executable(dsa_segment_tree_lab src/dsa/segment_tree.cpp)
add_executable(dsa_graph_algorithms_lab src/dsa/graph_algorithms.cpp)
add_executable(dsa_trie_lab src/dsa/trie.cpp)

enable_testing()
add_test(NAME sanity COMMAND sanity)
add_test(NAME oop_lab COMMAND oop_lab)
add_test(NAME raii_lab COMMAND raii_lab)
add_test(NAME lru_cache_lab COMMAND lru_cache_lab)
add_test(NAME thread_pool_lab COMMAND thread_pool_lab)
add_test(NAME order_book_lab COMMAND order_book_lab)
add_test(NAME dsa_disjoint_set_lab COMMAND dsa_disjoint_set_lab)
add_test(NAME dsa_segment_tree_lab COMMAND dsa_segment_tree_lab)
add_test(NAME dsa_graph_algorithms_lab COMMAND dsa_graph_algorithms_lab)
add_test(NAME dsa_trie_lab COMMAND dsa_trie_lab)
